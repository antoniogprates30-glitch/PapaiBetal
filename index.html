// --- ATUALIZAÇÃO DA LÓGICA DE SALDO --- function updateUI() { // Agora o saldo mostrado é o saldo real. // Não subtraímos a 'bet' visualmente antes da hora. document.getElementById('balance').innerText = Math.floor(balance); document.getElementById('bet-amount').innerText = bet; } function modBet(val) { // Só permite aumentar a aposta se o saldo for suficiente if (!isMinesActive && (balance >= bet + val)) { bet += val; updateUI(); } } /* LÓGICA DADOS ATUALIZADA */ function playDice() { if (!selectedOption) { document.getElementById('status-msg').innerText = "Escolha Par ou Ímpar!"; return; } if (bet <= 0 || balance < bet) return; const visual = document.getElementById('simple-visual'); visual.style.transform = "rotate(720deg)"; setTimeout(() => { visual.style.transform = "rotate(0deg)"; let res = Math.floor(Math.random() * 6) + 1; visual.innerText = (res === 1 ? '⚀' : res === 2 ? '⚁' : res === 3 ? '⚂' : res === 4 ? '⚃' : res === 5 ? '⚄' : '⚅'); let isEven = res % 2 === 0; let won = (selectedOption === 'Par' && isEven) || (selectedOption === 'Ímpar' && !isEven); if (won) { // Se ganhar, o saldo aumenta com o lucro (ex: 1.95x a aposta) let profit = bet * 0.95; balance += profit; document.getElementById('status-msg').innerText = `VENCEU! +$${profit.toFixed(2)}`; document.getElementById('status-msg').style.color = "var(--accent-gold)"; } else { // Se perder, a aposta é descontada agora balance -= bet; document.getElementById('status-msg').innerText = "PERDEU!"; document.getElementById('status-msg').style.color = "var(--primary-red)"; } bet = 0; // Reseta aposta após a rodada updateUI(); }, 600); } /* LÓGICA MINES ATUALIZADA */ function startMines() { if (bet <= 0 || balance < bet) { document.getElementById('status-msg').innerText = "Aposta Inválida!"; return; } // Note que NÃO descontamos o balance aqui. // O valor fica "preso" no jogo. isMinesActive = true; minesFound = 0; mineLocations = []; const numBombs = parseInt(document.getElementById('bomb-range').value); while (mineLocations.length < numBombs) { let r = Math.floor(Math.random() * 25); if (!mineLocations.includes(r)) mineLocations.push(r); } document.getElementById('main-action-btn').innerText = "CASH OUT"; document.getElementById('bomb-range').disabled = true; document.getElementById('status-msg').innerText = "BOA SORTE!"; document.getElementById('status-msg').style.color = "white"; } function cashOutMines() { if (minesFound === 0) return; let mult = calculateMinesMultiplier(); let profit = (bet * mult) - bet; // Calculamos apenas o lucro líquido em cima da aposta balance += profit; endMines(`GANHOU: $${(bet * mult).toFixed(2)}`, "var(--accent-gold)"); } function gameOverMines() { balance -= bet; // Perdeu? Agora o saldo diminui. endMines("BUM! VOCÊ PERDEU.", "var(--primary-red)"); }
